# 3.5 接口扩展中的泛型约束

[← 3. 接口扩展](index.md)

泛型接口扩展可用 `where` 同时约束被扩展类型的多个类型参数，约束满足时才获得接口能力。

```cangjie cjtest=compile id=verified-7d7d6ed68c-1
package constrained_interface_extension

class Pair<T1, T2> {
    let first: T1
    let second: T2

    public init(a: T1, b: T2) {
        first = a
        second = b
    }
}

interface Eq<T> {
    func equals(other: T): Bool
}

// 当 T1、T2 可判等时，让 Pair 实现 Eq 接口
extend<T1, T2> Pair<T1, T2> <: Eq<Pair<T1, T2>> where T1 <: Eq<T1>, T2 <: Eq<T2> {
    public func equals(other: Pair<T1, T2>): Bool {
        first.equals(other.first) && second.equals(other.second)
    }
}

class NumberBox <: Eq<NumberBox> {
    let value: Int64

    public init(value: Int64) {
        this.value = value
    }

    public func equals(other: NumberBox): Bool {
        value == other.value
    }
}

main(): Int64 {
    let left = Pair(NumberBox(1), NumberBox(2))
    let right = Pair(NumberBox(1), NumberBox(2))
    println(left.equals(right))
    return 0
}
```
