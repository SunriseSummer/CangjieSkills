# 2.2 访问成员

[← 2. 反射（动态特性）](index.md)

用 `ClassTypeInfo` 查询 `public` 成员；成员集合顺序不稳定，应按名称或注解选择，读值后用 `as T` 解包，写值前检查 `isMutable()`。

仅 `public` 成员对反射可见。`TypeInfo` 及其子类 `ClassTypeInfo` 提供以下访问接口：

- `ClassTypeInfo.instanceVariables` 只包含当前类声明的 `public` 实例变量，不包含继承成员，且集合遍历顺序不保证恒定；应按 `name` 选择成员，不能依赖数组下标。
- `getValue` 返回 `Any`，用 `as T` 得到 `Option<T>` 后再解包；`setValue` 前先用 `isMutable()` 检查可写性，并传入与 `typeInfo` 相符的值。
- 已知成员名时优先使用 `getInstanceVariable`、`getInstanceProperty` 等精确查找；需要按注解发现成员时再遍历集合并调用 `findAnnotation<T>()`。

### 访问变量
```cangjie cjtest=syntax id=syntax-67df6f922b-1 form=unit
import std.reflect.*

public class Foo {
    public static var param1 = 20
    public var param2 = 10
}

main() {
    let obj = Foo()
    let info = ClassTypeInfo.of(obj)
    // 访问静态变量
    let sv = info.getStaticVariable("param1")
    println((sv.getValue() as Int64).getOrThrow())  // 20
    sv.setValue(8)
    println((sv.getValue() as Int64).getOrThrow())  // 8
    // 访问实例变量
    let iv = info.getInstanceVariable("param2")
    println((iv.getValue(obj) as Int64).getOrThrow())  // 10
    iv.setValue(obj, 25)
    println((iv.getValue(obj) as Int64).getOrThrow())  // 25
}
```

### 访问属性
```cangjie cjtest=compile id=verified-67df6f922b-2
package reflect_property_example

import std.reflect.*

public class Profile {
    private var storedScore: Int64 = 7

    public prop name: String {
        get() { "Ada" }
    }

    public mut prop score: Int64 {
        get() { storedScore }
        set(value) { storedScore = value }
    }
}

main(): Int64 {
    let profile = Profile()
    let info = ClassTypeInfo.of(profile)
    let nameProperty = info.getInstanceProperty("name")
    let scoreProperty = info.getInstanceProperty("score")

    println((nameProperty.getValue(profile) as String).getOrThrow())
    if (scoreProperty.isMutable()) {
        scoreProperty.setValue(profile, 42)
    }
    println((scoreProperty.getValue(profile) as Int64).getOrThrow())
    return 0
}
```

### 调用函数
```cangjie cjtest=syntax id=syntax-67df6f922b-3 form=unit
import std.reflect.*

public class Calculator {
    public static func add(a: Int64, b: Int64): Int64 { a + b }
}

main() {
    let intInfo = TypeInfo.of<Int64>()
    let funcInfo = TypeInfo.of<Calculator>().getStaticFunction("add", intInfo, intInfo)
    let result = (funcInfo.apply(TypeInfo.of<Calculator>(), [1, 1]) as Int64).getOrThrow()
    println(result)  // 2
}
```

## 已验证示例

成员集合只用于发现候选项，不能依赖其顺序。下面按字段名选择成员、读取自定义注解，并在确认可写后通过反射赋值。

```cangjie cjtest=run id=language.reflection-field-lookup.run form=unit timeout=20s
package reflection_field_lookup_example

import std.reflect.*

@Annotation[target: [MemberVariable]]
public class ExternalName {
    public let value: String

    public const init(value: String) {
        this.value = value
    }
}

public class Profile {
    @ExternalName["display_name"]
    public var name: String = "Ada"

    public var age: Int64 = 42
}

main(): Unit {
    let profile = Profile()
    for (field in ClassTypeInfo.of(profile).instanceVariables) {
        if (field.name == "name") {
            let label = field.findAnnotation<ExternalName>().getOrThrow()
            if (field.isMutable()) {
                field.setValue(profile, "Grace")
            }
            let value = (field.getValue(profile) as String).getOrThrow()
            println("${label.value}:${value}")
        }
    }
}
```

