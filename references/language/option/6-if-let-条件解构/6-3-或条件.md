# 6.3 或条件（`||`）

[← 6. if-let 条件解构](index.md)

用 `||` 连接时，模式中**不能有变量绑定**，只能使用通配符 `_`
```cangjie cjtest=run id=language.option.if-let-or.run form=unit timeout=30s
package option_if_let_or_example

func hasValue(first: ?Int64, second: ?Int64): Bool {
    if (let Some(_) <- first || let Some(_) <- second) {
        true
    } else {
        false
    }
}

main(): Unit {
    let primary: ?Int64 = None
    let fallback: ?Int64 = Some(7)

    if (hasValue(primary, fallback)) {
        println("至少一个有值")
    }

    let empty: ?Int64 = None
    if (hasValue(primary, empty)) {
        println("不会执行")
    } else {
        println("两个都是 None")
    }
}
```

```text cjtest=expect for=language.option.if-let-or.run stream=stdout match=exact
至少一个有值
两个都是 None
```
